        function performSearch() {
            const query = document.getElementById('searchInput').value.trim().toUpperCase();
            const resultsContainer = document.getElementById('searchResults');
            const cardsContainer = document.getElementById('searchCardsContainer');
            cardsContainer.innerHTML = '';

            if (!query) return;

            const matches = inspectionRecords.filter(r => 
                r.partNo.toUpperCase().includes(query) || r.jigNo.toUpperCase().includes(query)
            );

            resultsContainer.classList.remove('hidden');

            if (matches.length === 0) {
                cardsContainer.innerHTML = `
                    <div class="col-span-2 text-center py-8 text-slate-400 glass-panel rounded-xl">
                        <i class="fa-solid fa-triangle-exclamation text-3xl text-amber-400 mb-2"></i>
                        <p>ไม่พบข้อมูลชิ้นงานสำหรับ "${query}" ในระบบ Database</p>
                    </div>
                `;
                return;
            }

            matches.forEach(rec => {
                const card = document.createElement('div');
                card.className = "bg-slate-800/80 border border-slate-700 rounded-xl p-5 hover:border-slate-500 transition space-y-3 shadow-lg";
                
                const isOk = rec.status === 'OK';
                const statusBadge = isOk 
                    ? `<span class="bg-emerald-500 text-slate-950 font-extrabold px-3 py-1 rounded-full text-xs">PASSED (OK)</span>`
                    : `<span class="bg-red-500 text-white font-extrabold px-3 py-1 rounded-full text-xs">FAILED (NG)</span>`;

                card.innerHTML = `
                    <div class="flex justify-between items-start">
                        <div>
                            <span class="text-xs text-slate-400">PART NUMBER:</span>
                            <div class="text-xl font-mono font-bold text-white">${rec.partNo}</div>
                        </div>
                        ${statusBadge}

                        <div>Jig ID: <span class="text-white">${rec.jigNo}</span></div>

                        <div class="col-span-2">เวลาบันทึก: <span class="text-slate-400">${rec.timestamp}</span></div>
                    </div>
                    <div class="flex justify-between items-center text-xs pt-1">
                        <span class="text-slate-400">สรุปการวัด: <strong class="text-emerald-400">${rec.okCount} OK</strong>, <strong class="text-red-400">${rec.ngCount} NG</strong></span>

        function openRealtimeModal(robotId) {
            const rec = inspectionRecords.find(r => r.robot === robotId) || inspectionRecords[0];

        function openModalWithRecord(record) {
            currentRecordInModal = record;
            document.getElementById('mPartNo').textContent = record.partNo;
            document.getElementById('mJigNo').textContent = record.jigNo;
            document.getElementById('mRobot').textContent = record.robot;
            document.getElementById('mTimestamp').textContent = record.timestamp;

            const badge = document.getElementById('mOverallBadge');
            if (record.status === 'OK') {
                badge.className = "px-3 py-1 text-xs font-bold rounded-full bg-emerald-500 text-slate-950";
                badge.textContent = "PASSED (OK)";
            } else {
                badge.className = "px-3 py-1 text-xs font-bold rounded-full bg-red-500 text-white";
                badge.textContent = "FAILED (NG)";
            }

            renderPointsGrid(record.points);
            selectPoint(1);

            document.getElementById('detailModal').classList.remove('hidden');
        }

        function renderPointsGrid(points) {
            const grid = document.getElementById('pointsGrid');
            grid.innerHTML = '';

            points.forEach(p => {
                const btn = document.createElement('button');
                btn.id = `pt-btn-${p.id}`;

        function selectPoint(pointId) {
            activeSelectedPoint = pointId;
            if (!currentRecordInModal) return;

            const pointData = currentRecordInModal.points.find(p => p.id === pointId);
            const pointDef = pointDefinitions.find(d => d.id === pointId) || pointDefinitions[0];

            document.querySelectorAll('#pointsGrid button').forEach(btn => btn.classList.remove('ring-2', 'ring-blue-400', 'scale-105'));
            const activeBtn = document.getElementById(`pt-btn-${pointId}`);
            if (activeBtn) activeBtn.classList.add('ring-2', 'ring-blue-400', 'scale-105');

            document.getElementById('pdTitle').textContent = `Point #${pointData.id} : ${pointData.name}`;
            document.getElementById('pdNominal').textContent = `${pointData.nominal.toFixed(3)} mm`;
            document.getElementById('pdTolerance').textContent = `± ${pointData.tol.toFixed(3)} mm`;

            const measuredEl = document.getElementById('pdMeasured');
            measuredEl.textContent = `${pointData.measured.toFixed(3)} mm`;

            const statusEl = document.getElementById('pdStatus');
            if (pointData.status === 'OK') {
                statusEl.className = "px-2.5 py-0.5 rounded text-xs font-bold bg-emerald-500/20 text-emerald-400 border border-emerald-500/30";
                statusEl.textContent = "OK (ผ่านเกณฑ์)";
                measuredEl.className = "font-mono text-xl font-extrabold text-emerald-400";
            } else {
                statusEl.className = "px-2.5 py-0.5 rounded text-xs font-bold bg-red-500/20 text-red-400 border border-red-500/30";
                statusEl.textContent = "NG (ตกเกณฑ์)";
                measuredEl.className = "font-mono text-xl font-extrabold text-red-400";
            }

            const targetDot = document.getElementById('svgTargetDot');
            const targetCircle = document.getElementById('svgTargetCircle');
            targetDot.setAttribute('cx', pointDef.cx);
            targetDot.setAttribute('cy', pointDef.cy);
            targetCircle.setAttribute('cx', pointDef.cx);
            targetCircle.setAttribute('cy', pointDef.cy);
        }

        function exportToCSV() {

            csvContent += "Timestamp,Part No,Jig No,Robot Station,OK Points,NG Points,Overall Status\n";

            inspectionRecords.forEach(r => {
                csvContent += `${r.timestamp},${r.partNo},${r.jigNo},${r.robot},${r.okCount},${r.ngCount},${r.status}\n`;
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `ISUZU_KNUCKLE_Inspection_${new Date().toISOString().slice(0,10)}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }
